\item Given the importance of derivative in a wide range of settings, there are many different types of derivatives as well as a variety of different notations used to the represent them, including:
\[
f'(\vecnot{x}),  \qquad \mathsf{D} f(\vecnot{x}) , \qquad J(\vecnot{x}), \qquad \dot{f}(\vecnot{x}) , \qquad  \frac{ d \vecnot{y}}{ d \vecnot{x}}, \qquad \nabla f(\vecnot{x})

\item For a function $f \colon X \to Y$, it is best to view $f'(x)$ as a \emph{linear transform} $T \colon X \to Y$ from the domain to codomain.
This transform maps an infinitesimal input perturbation $\vecnot{h}$ to an infinitesimal output perturbation:
\[
f'(\vecnot{x})(\vecnot{h}) = T \vecnot{h} = J(\vecnot{x}) \vecnot{h}
\]
 
 
 \item We will see that the definition of a derivative used in multivariable calculus  extends naturally to a function between two  \emph{Banach spaces:}
 \begin{itemize}
 \item The linear structure of vectors spaces defines differences
 \item The norms $\| \cdot\|$ provide the notion length and the topology of the induced metric space
 \item Completeness ensures that a limit point of sequence remains in the space.
 \end{itemize}
 
 
 \item To motivate the definition, recall that a function $f$ from from one metric space to another  is \textbf{continuous} at $\vecnot{x}$ if and only if $ f(\vecnot{x}_n) \to f(\vecnot{x})$ for every converging sequence  $\vecnot{x}_n \to \vecnot{x}$.  
% \[
% f(\vecnot{x}_n) \to f(\vecnot{x}) \quad \text{for every converging sequence  $\vecnot{x}_n \to \vecnot{x}$}
% \]
This condition can also be expressed as 
\[
\lim_{\vecnot{y} \to \vecnot{x} }f( \vecnot{y} ) =  f(\vecnot{x}) 
\]
where the notation $\vecnot{y} \to \vecnot{x}$ means that the limit holds for \emph{for every converging sequence}. 
 
 
\item \textbf{Example:}  Consider the Euclidean space $X = \bbR^2$ and let $f \colon X \to \bbR$ be defined by 
\begin{align*}
f(x_1 , x_2) = \begin{dcases} 
1 & x_1 > 0, x_1 = x_2^2\\
0 & \text{otherwise}
\end{dcases}
\end{align*}
Is this function continuous at $\vecnot{0}$?
\begin{itemize}
\item The limit holds for many converging sequences. Indeed, we see that
\[
\lim_{t \to 0^+} f(t \vecnot{x} )  = \vecnot{0} \quad \text{for every $\vecnot{x} \in \bbR^2$}
\]
\item However, there exists at least one converging sequence for which it fails. Namely, if
\[
\vecnot{x}_n = \left ( \tfrac{1}{n^2}, \; \tfrac{1}{n} \right )%^\top = \begin{bmatrix} 1/n^2  \\  1/n  \end{bmatrix} , \qquad n \in \bbN
\]
Then, $\|\vecnot{x}_n \| \to0$  but  $f(\vecnot{x}_n) = 1$ for all $n \in \bbN$ and thus
\[
\lim_{n \to \infty} f(\vecnot{x}_n)  = 1  \ne 0 = f( \lim_{n \to \infty} \vecnot{x}_n) %= f( \vecnot{0}) 
\]
\end{itemize}

\item \textbf{Chain Rule:}  Let $X,Y,Z$ be Banach spaces and let $f \colon X \rightarrow Y$ and $g\colon Y \to Z$ be functions.
If $f$ is Fr\'{e}chet differentiable at $\vecnot{x}$ and $g$ is Fr\'{e}chet differentiable at $\vecnot{y}=f(\vecnot{x})$, then the composition $\phi = g \circ f$ defined by 
\[
\phi (\vecnot{x}) = g(f(\vecnot{x}))
\]
is Fr\'{e}chet differentiable at $\vecnot{x}$ with derivative
\[
\phi' (\vecnot{x}) =  g'(f(\vecnot{x}))\circ f'(\vecnot{x})
\]


\item \textbf{Lemma:} Let $X,Y$ be Banach spaces and $f \colon X \rightarrow Y$ be a function.
If the Fr\'{e}chet derivative $f'(\vecnot{x})$ exists and satisfies $\| f'(\vecnot{x})\|_{\mathrm{op}} \leq L$ for all $\vecnot{x}$ in a convex set $A\subseteq X$, then $f$ is Lipschitz continuous on $A$ with Lipschitz constant $L$.

\item \textbf{Proof Sketch:} 
\begin{itemize}
\item For simplicity, we will suppose $X = Y = \bbR^n$. % and let $J(\vecnot{x})$ be the Jacobian matrix of $f$
 Fix any $\vecnot{x}, \vecnot{y} \in A$ and set $\vecnot{h} = \vecnot{y} - \vecnot{x}$
\item Define  $\phi \colon [0,1] \to X$ according to 
\[
\phi(t) =f (  \vecnot{x} + t  \vecnot{h} ) 
\]
such that $\phi(0) = \vecnot{x}$ and $\phi(1) = \vecnot{y}$. 
\item Since $A$ is convex, $\vecnot{x} + t  \vecnot{h}\in A$ for all $t \in [0,1]$. By the second fundamental theorem of calculus and the chain rule we can write
\begin{align*}
f(\vecnot{y}) - f( \vecnot{x}) & =\phi(1)  - \phi(0)   = \int_0^1J ( \vecnot{x} + t \vecnot{h})  \vecnot{h} \, dt %\\
%& = \int_0^1  g'(f(\vecnot{x})) f'(\vecnot{x})
\end{align*}
\item 
Talking the norm of both sides leads to 
\begin{alignat*}{3}
\| f(\vecnot{y}) - f( \vecnot{x}) \|& =  \left\|  \int_0^1J ( \vecnot{x} + t \vecnot{h})  \vecnot{h} \, dt  \right\|  \\
& \le   \int_0^1 \left\| J ( \vecnot{x} + t \vecnot{h})   \vecnot{h}\right\|  \, dt &\quad & \text{Jensen's inequality}\\
& \le   \int_0^1 \underbrace{\|  J ( \vecnot{x} + t \vecnot{h}) \|_{op} }_{\le L}    \| \vecnot{h} \|   \, dt\\
& \le L\| \vecnot{y} - \vecnot{x}\| % && \text{ $\|  J ( \vecnot{x} ) \|_{op} \le L$ for all  $\vecnot{x} \in A$}
%& = \int_0^1  g'(f(\vecnot{x})) f'(\vecnot{x})
\end{alignat*}
\end{itemize}

\end{itemize}


\section{G\^{a}teaux derivative}
\begin{itemize}


\item   Next, we introduce a different notion of a derivative which  can be applied when the domain of the function is a vector space but not necessarily a Banach space. The basic idea is that that the metric topology of the field  (i.e., the real or complex numbers) provides a of convergence with respect to a given \emph{direction}. For simplicity, these notes consider only the case of a vector space over the reals. 

\item \textbf{Definition:} Let $f $ be a mapping from a real vector space $X$ to a Banach space $(Y,\|\cdot\|)$.
Then, if it exists, the \textbf{directional derivative} of $f$ in direction $\vecnot{h}$  is given by 
% \textcolor{black} d of $f$ at  $\vecnot{x}$ in direction $\vecnot{h}$ is given by 
\[
f'(\vecnot{x};\vecnot{h})  \coloneqq \lim_{t \to 0^+} \frac{f(\vecnot{x}+t \vecnot{h}) - f(\vecnot{x})}{t}
\]
where $t \to 0^+$ means the limit is taken along any sequence of strictly positive numbers that converges to $0$ in the standard topology of the real numbers. 

\item \textbf{Example:}  Consider $X =Y=\bbR^2$ and $f(\vecnot{x}) = (x_1 x_2,x_1+x_2^2)$.
For $\vecnot{x}=(1,1)$, $\vecnot{h}=(1,2)$: 
\[ 
f' ( \vecnot{x}; \vecnot{h} ) = \frac{d}{d t} ((1+t)(1+2t),(1+t)+(1+2t)^2) \Big|_{t=0} = (3,5)
\]



\item \textbf{Definition:} Let $f $ be a mapping from a real vector space $X$ to a Banach space $(Y,\|\cdot\|)$. Then, $f$ is \textbf{G\^{a}teaux differentiable} at $\vecnot{x}$ if the directional derivative exists for all $\vecnot{h} \in X$ and there exists a  continuous linear transformation $T\colon X \to Y$ satisfying

%\end{equation*}
%where the limit $\vecnot{h} \to \vecnot{0}$ is with respect to the induced metric on $X$.

\section{Gradient descent} 

\begin{itemize}
\item In multivariate calculus, \textbf{gradient} of a differentiable function $f\colon \bbR^n \to \bbR$ %with $m=1$, the Jacobian is related to the \textcolor{blue}{gradient}
is given by 
\[
 \nabla f(\vecnot{x}) \coloneqq  f'(\vecnot{x})^\top  = \left[ \begin{array}{cccc} \frac{\partial  f}{\partial  x_1} (\vecnot{x}) & \frac{\partial  f}{\partial  x_2} (\vecnot{x}) & \cdots & \frac{\partial  f}{\partial  x_n} (\vecnot{x}) \end{array} \right]^\top
  \]
It is worth noting that the orientation of the gradient vector (i.e., row versus column vector) is sometimes defined differently.
This is because derivatives can be understood as linear transforms and either orientation can be used to define the correct linear transform. 

\item \textbf{Example:} Let $X$ be a Hilbert space over $\bbR$ and $f\colon X \to \bbR$ be a real functional.
If the Fr\'{e}chet derivative $f'(\vecnot{x})$ exists, then it is a continuous linear functional on $X$.
Thus, the Riesz representation theorem guarantees that there is a vector $\vecnot{u} \in X$ such that
$ f'(\vecnot{x})(\vecnot{h}) = \tinner{\vecnot{h} }{ \vecnot{u} } $ for all $\vecnot{h}\in X$.
This vector is called the gradient $\nabla f(\vecnot{x})$ and it follows that
\[
 f'(\vecnot{x})(\vecnot{h}) = \tinner{\vecnot{h} }{ \nabla f (\vecnot{x}) } \quad  \text{ for all } \vecnot{h}\in X
\]

\item \textbf{Definition:} [Gradient Descent]
Let $f \colon X \rightarrow \bbR$ be a mapping from a Hilbert space $X$ to the standard Banach space of real numbers. Starting from $\vecnot{x}_1 \in X$, \textbf{gradient descent} defines the sequence
\[ \vecnot{x}_{n+1} = \vecnot{x}_n - \delta_n \nabla f(\vecnot{x}_n), \]
where $\delta_n$ is the step size for the $n$-th step.


\item \textbf{Lemma:} Let $f \colon X \rightarrow \bbR$ map the Hilbert space $X$ to the real numbers.
If $\nabla f(\vecnot{x})$ exists and satisfies $ \| \nabla f(\vecnot{y}) - \nabla f(\vecnot{x}) \| \leq L \| \vecnot{y} - \vecnot{x} \|$, then
\[
 \big| f(\vecnot{y}) - f(\vecnot{x}) - \tinner{\vecnot{y}-\vecnot{x} }{ \nabla f (\vecnot{x}) } \big| \leq \frac{1}{2} L \| \vecnot{y}-\vecnot{x} \|^2
\]


\item \textbf{Proof:} 
%\begin{itemize}
Let $\vecnot{h} = \vecnot{y}-\vecnot{x}$ and $\phi(t) = f(\vecnot{x} + t \vecnot{h})$. Then, $\phi'(t) = \tinner{ \vecnot{h} }{ \nabla f(\vecnot{x} + t \vecnot{h}) }$ and 
\begin{align*}
\big| f(\vecnot{y}) -  f(\vecnot{x}) - \tinner{ \vecnot{h} }{ \nabla f(\vecnot{x}) } \big|  &= \left| \int_{0}^1 \left( \phi'(t) - \phi'(0) \right) \dd t \right| \\
&=  \left| \int_{0}^1 \tinner{ \vecnot{h} }{ \nabla f(\vecnot{x} + t \vecnot{h}) - \nabla f(\vecnot{x}) } \, \mathrm{d}t \right| \\
&\leq \left| \int_{0}^1  \| \vecnot{h} \| \, \| \nabla f(\vecnot{x} + t \vecnot{h}) - \nabla f(\vecnot{x}) \| \, \mathrm{d}t \right| \\
&\leq \int_{0}^1 \|\vecnot{h} \| L \| t \vecnot{h} \| \, \mathrm{d}t \\
& = \frac{1}{2} L \| \vecnot{h} \|^2
\end{align*}
